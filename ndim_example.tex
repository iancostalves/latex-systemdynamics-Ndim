% Example of N-dimensional stocks in a system dynamics diagram.
% Stocks can be differentiated by pathway (p), technology (t) or any other
% discrete dimension.  The \ndstock command draws three stacked boxes and
% places a small dimension identifier either above or below the stack.
\documentclass{standalone}
\pagestyle{empty}
\usepackage{systemdynamics}

\begin{document}
\begin{tikzpicture}

% ── Ordinary single-dimension stock ──────────────────────────────────────────
\node (energy) [stock] {Energy};
\flowIn{energy}{}
\flowOut{energy}{}

% ── Stock indexed over pathways (p), label shown below ────────────────────────
% Usage: \ndstock[tikz-options]{name}{label}{dimtype}{dimpos}
\ndstock[below=8em of energy]{energy_p}{Energy}{$p$}{below}
\flowIn{energy_p}{}
\flowOut{energy_p}{}
% Arrows can reference the main node as usual
\sdarrow{energy}{energy_p}{}

% ── Stock indexed over technologies (t), label shown above ───────────────────
\ndstock[right=14em of energy_p]{energy_t}{Energy}{$t$}{above}
\flowIn{energy_t}{}
\flowOut{energy_t}{}

% ── Stock indexed over layers (a), label shown below ─────────────────────────
\ndstock[right=14em of energy]{cap}{Capacity}{$a$}{below}
\flowIn{cap}{}
\flowOut{cap}{}
\sdarrow{cap}{energy_t}{bend right=20}

\end{tikzpicture}
\end{document}
